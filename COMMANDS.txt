# Exported model val inference, bbox AP, overlays (CoreML / TFLite)
# Analogous to mmpose/mmdeploy validation pipeline.
#
# Conda envs:
#   mmdeploy         TorchScript export (tools/deploy.py)
#   model_conversion CoreML/TFLite conversion (see mmdeploy/COMMANDS.txt)
#   mmdet            exported val inference + bbox AP + overlays (this section)
#
# One-time for TFLite val/overlays in mmdet env:
#   pip install "ai-edge-litert==2.1.5"
# Do NOT install tensorflow into mmdet (would force numpy upgrade).
#
# Scripts (under mmdetection/tools/):
#   exported_export_val_predictions.py  run exported models on val, write COCO detections JSON
#   exported_compare_ap.py              compare bbox AP between two exported prediction JSONs
#   overlay_val_predictions.py          render val overlay MP4 from detections JSON
#
# Paths / naming (relative to mmdetection/):
#   Val annotations:  data/${VAL_DATA_DIRNAME}/annotations/person_keypoints_val2017.json
#   Val images:       data/${VAL_DATA_DIRNAME}/val2017/
#   Predictions:      predictions/{coreml|tflite}_{model_stem}_{dataset}.detections.json
#   AP report:        reports/ap_exported.txt
#   Overlay videos:   overlays/{predictions_filename}.mp4 (10 fps)
##################################################################################

Export Model (TorchScript -> CoreML/TFLite):
# Full conversion steps live in mmdeploy/COMMANDS.txt ("To convert rtmdet to CoreML").
# Summary:
# 1. Env mmdeploy — TorchScript export:
cd ~/visionAI/mmdeploy
conda activate mmdeploy
export PYTHONPATH=$(pwd):$PYTHONPATH
python tools/deploy.py \
    configs/mmdet/detection/detection_coreml_static-320.py \
    ../mmdetection/configs/rtmdet/rtmdet_nano_320_8xb32_coco_finetune.py \
    ../mmdetection/work_dirs/rtmdet_nano_merged_gbe_v10_gbe_app_vids_v5/best_coco_bbox_mAP_epoch_*.pth \
    ../mmdetection/data/merged_gbe_v10_gbe_app_vids_v5/val2017/$(ls ../mmdetection/data/merged_gbe_v10_gbe_app_vids_v5/val2017 | head -1) \
    --work-dir work_dir/${MODEL_NAME} \
    --device cpu \
    --dump-info

# 2. Env model_conversion — CoreML:
conda deactivate && conda activate model_conversion
cd ~/visionAI/mmdeploy
# Update model path in tools/rtmdet_torch_to_coreml.py if needed, then:
python tools/rtmdet_torch_to_coreml.py

# 3. Env model_conversion — TFLite (optional):
# See mmdeploy/COMMANDS.txt TFLite steps (torchscript_to_onnx.py, onnx2tf, tf_to_tflite.py)

# Exported val inference (CoreML and/or TFLite):
cd ~/visionAI/mmdetection

    --img-prefix ${VAL_IMG}
